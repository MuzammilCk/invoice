from reportlab.pdfgen import canvas
from reportlab.lib.pagesizes import A4
from reportlab.pdfbase import pdfmetrics
from reportlab.pdfbase.ttfonts import TTFont
from reportlab.lib.colors import HexColor, white

# ---------- Invoice data (edit for each new invoice) ----------
BUSINESS_NAME = "VIVA ELECTRONICS"
BUSINESS_LINES = ["Chettiyamkinar, Kottakkal, Malappuram, Kerala",
                  "9995880059  ·  9633334786"]
DATE = "10 November 2026"
BILL_TO = "Illikkal Auditorium"
ITEMS = [("Sub Amp Servicing", 900.00)]
OUT = "invoice.pdf"

# ---------- Fonts ----------
FONT_DIR = "./"  # folder containing Poppins-Regular.ttf, Poppins-Medium.ttf, Poppins-Bold.ttf
pdfmetrics.registerFont(TTFont("P", FONT_DIR + "Poppins-Regular.ttf"))
pdfmetrics.registerFont(TTFont("PM", FONT_DIR + "Poppins-Medium.ttf"))
pdfmetrics.registerFont(TTFont("PB", FONT_DIR + "Poppins-Bold.ttf"))

# ---------- One colour system: ink + neutrals ----------
INK = HexColor("#111111")      # text, table header, total bar
GRAY = HexColor("#6B6B6B")     # secondary text
SOFT = HexColor("#A3A3A3")     # secondary text on black
LINE = HexColor("#E6E6E6")     # hairlines

W, H = A4
M = 48          # page margin (8pt grid)
R = 8           # one corner radius, used everywhere


def spaced(c, x, y, text, font, size, space, align="left"):
    width = pdfmetrics.stringWidth(text, font, size) + space * len(text)
    if align == "right":
        x -= width
    t = c.beginText(x, y)
    t.setFont(font, size)
    t.setCharSpace(space)
    t.textOut(text)
    t.setCharSpace(0)
    c.drawText(t)


def money(v):
    """Rupee amount with Indian digit grouping, e.g. 125000 -> ₹1,25,000.00"""
    whole, frac = f"{v:.2f}".split(".")
    if len(whole) > 3:
        head, tail, parts = whole[:-3], whole[-3:], []
        while len(head) > 2:
            parts.insert(0, head[-2:])
            head = head[:-2]
        if head:
            parts.insert(0, head)
        whole = ",".join(parts + [tail])
    return f"\u20b9{whole}.{frac}"


c = canvas.Canvas(OUT, pagesize=A4)
c.setTitle(f"Invoice - {BILL_TO}")
c.setAuthor("Viva Electronics")

# --- Business block, top-left ---
c.setFillColor(INK)
c.setFont("PB", 20)
c.drawString(M, H - 72, BUSINESS_NAME)
c.setFillColor(GRAY)
c.setFont("P", 9.5)
y = H - 92
for line in BUSINESS_LINES:
    c.drawString(M, y, line)
    y -= 14

# --- Date, top-right (aligned to name baseline and first detail line) ---
c.setFillColor(GRAY)
spaced(c, W - M, H - 72, "DATE", "PM", 7.5, 1.4, align="right")
c.setFillColor(INK)
c.setFont("PM", 10.5)
c.drawRightString(W - M, H - 92, DATE)

# --- Hairline divider ---
c.setStrokeColor(LINE)
c.setLineWidth(0.8)
c.line(M, H - 128, W - M, H - 128)

# --- Billed to ---
c.setFillColor(GRAY)
spaced(c, M, H - 168, "BILLED TO", "PM", 7.5, 1.4)
c.setFillColor(INK)
c.setFont("PM", 14)
c.drawString(M, H - 190, BILL_TO)

# --- Table ---
tx, tw = M, W - 2 * M
col_no, col_item, col_price = tx + 16, tx + 64, tx + tw - 16
top, hh, rh = H - 240, 32, 48

c.setFillColor(INK)
c.roundRect(tx, top - hh, tw, hh, R, fill=1, stroke=0)
c.setFillColor(white)
spaced(c, col_no, top - 20, "NO.", "PM", 7.5, 1.4)
spaced(c, col_item, top - 20, "ITEM", "PM", 7.5, 1.4)
spaced(c, col_price, top - 20, "PRICE", "PM", 7.5, 1.4, align="right")

y = top - hh
for i, (name, price) in enumerate(ITEMS, 1):
    base = y - rh / 2 - 4
    c.setFillColor(GRAY)
    c.setFont("P", 10)
    c.drawString(col_no, base, f"{i:02d}")
    c.setFillColor(INK)
    c.setFont("PM", 11)
    c.drawString(col_item, base, name)
    c.drawRightString(col_price, base, money(price))
    y -= rh
    c.setStrokeColor(LINE)
    c.setLineWidth(0.8)
    c.line(tx, y, tx + tw, y)

# --- Total: same black, same radius as the header; amount aligns with PRICE column ---
total = sum(p for _, p in ITEMS)
bw, bh = 248, 48
bx, by = tx + tw - bw, y - 24 - bh
c.setFillColor(INK)
c.roundRect(bx, by, bw, bh, R, fill=1, stroke=0)
base = by + bh / 2 - 6
c.setFillColor(SOFT)
spaced(c, bx + 20, base, "TOTAL", "PM", 7.5, 1.4)
c.setFillColor(white)
c.setFont("PB", 18)
c.drawRightString(col_price, base, money(total))

# --- Footer ---
c.setStrokeColor(LINE)
c.setLineWidth(0.8)
c.line(M, 72, W - M, 72)
c.setFillColor(GRAY)
c.setFont("P", 8.5)
c.drawCentredString(W / 2, 52, "Thank you for your business.")

c.showPage()
c.save()
print("saved", OUT)

from reportlab.pdfgen import canvas
from reportlab.lib.pagesizes import A4
from reportlab.pdfbase import pdfmetrics
from reportlab.pdfbase.ttfonts import TTFont
from reportlab.lib.colors import HexColor, white

# ---------- Invoice data (edit for each new invoice) ----------
NAME_1, NAME_2 = "VIVA", "ELECTRONICS"
BUSINESS_LINES = ["Chettiyamkinar, Kottakkal, Malappuram, Kerala",
                  "9995880059  ·  9633334786"]
DATE = "10 November 2026"
BILL_TO = "Illikkal Auditorium"
ITEMS = [("Sub Amp Servicing", 900.00)]          # (item name, price in rupees)
FOOTER = "Thank you for your business."
OUT = "invoice.pdf"

# ---------- Fonts: Poppins Regular / Medium / Bold (free, Google Fonts) ----------
FONT_DIR = "./"
pdfmetrics.registerFont(TTFont("P", FONT_DIR + "Poppins-Regular.ttf"))
pdfmetrics.registerFont(TTFont("PM", FONT_DIR + "Poppins-Medium.ttf"))
pdfmetrics.registerFont(TTFont("PB", FONT_DIR + "Poppins-Bold.ttf"))

# ---------- Palette ----------
INK, BLUE = HexColor("#0D0F1A"), HexColor("#2F3BFF")
GRAY, RULE = HexColor("#555A6B"), HexColor("#D9DBE6")

# ---------- Type scale (pt) ----------
BODY, LBL, ROW, DATE_V, REC = 10, 8.5, 12.5, 11.5, 19
WORD_V, WORD_E, TOTAL = 56, 34, 32
LEAD, ROW_H, LIMIT = 15, 56, 712

W, H = A4
M = 48


# ---------- helpers (all y values are measured DOWN from the top edge) ----------
def tw(s, font, size, hs=100, cs=0):
    return pdfmetrics.stringWidth(s, font, size) * hs / 100 + cs * len(s)


def fit(s, font, target, hs=100):
    return target / (pdfmetrics.stringWidth(s, font, 1) * hs / 100)


def text(c, x, y, s, font, size, color, anchor="l", hs=100, cs=0, outline=None, lw=1.0):
    w = tw(s, font, size, hs, cs)
    if anchor == "r":
        x -= w
    t = c.beginText(x, H - y)
    t.setFont(font, size)
    t.setHorizScale(hs)            # percent: 68 = squeezed to 68% width
    t.setCharSpace(cs)
    if outline is not None:        # outline-only text
        c.setStrokeColor(outline)
        c.setLineWidth(lw)
        t.setTextRenderMode(1)
    else:
        c.setFillColor(color)
    t.textOut(s)
    t.setTextRenderMode(0)
    t.setCharSpace(0)
    t.setHorizScale(100)
    c.drawText(t)


def label(c, x, y, s, color, anchor="l"):
    text(c, x, y, s, "PM", LBL, color, anchor=anchor, cs=1.6)


def poly(c, pts, fill):
    p = c.beginPath()
    p.moveTo(pts[0][0], H - pts[0][1])
    for x, y in pts[1:]:
        p.lineTo(x, H - y)
    p.close()
    c.setFillColor(fill)
    c.drawPath(p, fill=1, stroke=0)


def hline(c, y, color, lw=0.8):
    c.setStrokeColor(color)
    c.setLineWidth(lw)
    c.line(M, H - y, W - M, H - y)


def money(v):
    """Rupees with Indian digit grouping: 125000 -> ₹1,25,000.00"""
    whole, frac = f"{v:.2f}".split(".")
    if len(whole) > 3:
        head, tail, parts = whole[:-3], whole[-3:], []
        while len(head) > 2:
            parts.insert(0, head[-2:])
            head = head[:-2]
        if head:
            parts.insert(0, head)
        whole = ",".join(parts + [tail])
    return f"\u20b9{whole}.{frac}"


def wrap(s, font, size, maxw):
    lines, cur = [], ""
    for w in s.split():
        t = (cur + " " + w).strip()
        if tw(t, font, size) <= maxw:
            cur = t
        else:
            if cur:
                lines.append(cur)
            cur = w
    if cur:
        lines.append(cur)
    return lines or [s]


def leader(c, x0, x1, y, color, pitch=5.5, r=1.0):
    c.setFillColor(color)
    x = x0
    while x <= x1:
        c.circle(x, H - y, r, fill=1, stroke=0)
        x += pitch


# ---------- page parts ----------
def first_page_header(c):
    bV = M + 0.70 * WORD_V
    text(c, M - 1.5, bV, NAME_1, "PB", WORD_V, INK, hs=68)
    bE = bV + 10 + 0.70 * WORD_E
    text(c, M - 0.5, bE, NAME_2, "PB", WORD_E, INK, hs=80, cs=0.9, outline=INK, lw=0.8)
    y = bE + 28
    for line in BUSINESS_LINES:
        text(c, M, y, line, "P", BODY, GRAY)
        y += 15

    cw_, ch_, n = 148, 52, 8                       # date chip, notched corners
    cx = W - M - cw_
    poly(c, [(cx + n, M), (W - M, M), (W - M, M + ch_ - n), (W - M - n, M + ch_), (cx, M + ch_), (cx, M + n)], BLUE)
    label(c, cx + 16, M + 21, "DATE", white)
    text(c, cx + 16, M + 40, DATE, "PM", DATE_V, white)

    ry = 232                                        # recipient
    label(c, M, ry, "BILLED TO", BLUE)
    name = BILL_TO.upper()
    sR = min(REC, fit(name, "PB", 330, 84))
    text(c, M - 0.5, ry + 14 + 0.70 * sR, name, "PB", sR, INK, hs=84)


def table_header(c, ty):
    hline(c, ty, INK, 2)
    label(c, M, ty + 21, "NO.", BLUE)
    label(c, M + 64, ty + 21, "ITEM", BLUE)
    label(c, W - M, ty + 21, "PRICE", BLUE, anchor="r")
    y = ty + 34
    hline(c, y, RULE)
    return y


def footer(c):
    hline(c, H - 66, RULE)
    text(c, M, H - 42, FOOTER, "P", 9, GRAY)


def build():
    c = canvas.Canvas(OUT, pagesize=A4)
    c.setTitle(f"Invoice - {BILL_TO}")
    c.setAuthor("Viva Electronics")
    first_page_header(c)
    y = table_header(c, 312)

    for i, (item, price) in enumerate(ITEMS, 1):
        pw = tw(money(price), "PM", ROW)
        lines = wrap(item, "PM", ROW, (W - M) - pw - 28 - (M + 64))
        n = len(lines)
        rh = max(ROW_H, 40 + LEAD * (n - 1))
        if y + rh > LIMIT:                          # continue on a new page
            hline(c, y, INK, 2)
            footer(c)
            c.showPage()
            text(c, M - 0.5, M + 14, f"{NAME_1} {NAME_2}", "PB", 15, INK, hs=76)
            y = table_header(c, M + 36)
        mid = y + rh / 2
        text(c, M - 1, mid + 12, f"{i:02d}", "PB", 30, INK, hs=70, outline=INK, lw=0.9)
        fb = mid + 5 - LEAD * (n - 1) / 2
        for k, ln in enumerate(lines):
            text(c, M + 64, fb + LEAD * k, ln, "PM", ROW, INK)
        text(c, W - M, mid + 5, money(price), "PM", ROW, INK, anchor="r")
        y += rh
        last = i == len(ITEMS)
        hline(c, y, INK if last else RULE, 2 if last else 0.8)

    # total row: tag on the baseline, dotted leader, amount aligned with the PRICE column
    amt = money(sum(p for _, p in ITEMS))
    base = y + 30 + 0.70 * TOTAL
    aw = tw(amt, "PB", TOTAL, 88)
    cwid, chei = 74, 22
    cy0 = base - chei
    poly(c, [(M + 6, cy0), (M + cwid, cy0), (M + cwid, base - 6), (M + cwid - 6, base),
             (M, base), (M, cy0 + 6)], INK)
    label(c, M + 11, cy0 + 15, "TOTAL", white)
    leader(c, M + cwid + 12, (W - M) - aw - 14, base - chei / 2, BLUE)
    text(c, W - M, base, amt, "PB", TOTAL, BLUE, anchor="r", hs=88)

    footer(c)
    c.showPage()
    c.save()
    print("saved", OUT)


build()

from reportlab.pdfgen import canvas
from reportlab.lib.pagesizes import A4
from reportlab.pdfbase import pdfmetrics
from reportlab.pdfbase.ttfonts import TTFont
from reportlab.lib.colors import HexColor, white

# ---------- Invoice data (existing content only) ----------
NAME = "VIVA ELECTRONICS"
BUSINESS_LINES = ["Chettiyamkinar, Kottakkal, Malappuram, Kerala",
                  "9995880059  ·  9633334786"]
DATE = "10 November 2026"
BILL_TO = "Illikkal Auditorium"
ITEMS = [("Sub Amp Servicing", 900.00)]
FOOTER = "Thank you for your business."
OUT = "invoice.pdf"

# ---------- Fonts (free): FreeSans + FreeSans Bold (Helvetica clone, has the rupee sign),
# ---------- DejaVu Sans Mono Bold (readout). Put the three .ttf files next to this script. ----------
FONT_DIR = "./"
pdfmetrics.registerFont(TTFont("S", FONT_DIR + "FreeSans.ttf"))
pdfmetrics.registerFont(TTFont("SB", FONT_DIR + "FreeSansBold.ttf"))
pdfmetrics.registerFont(TTFont("M", FONT_DIR + "DejaVuSansMono-Bold.ttf"))

# ---------- Palette: warm greys + charcoal, ONE functional accent ----------
MOD, GRAY, INK = HexColor("#F1F0EC"), HexColor("#5E5C56"), HexColor("#1B1B1B")
TICK, HAIR = HexColor("#BDBBB3"), HexColor("#DAD8D1")
CHAR, BEZEL, DLEG = HexColor("#1E1E1D"), HexColor("#3C3B38"), HexColor("#A8A59C")
ORANGE = HexColor("#F2681A")

# ---------- Type scale (pt) ----------
LBL, BODY, ROW, DATE_V, REC, WORD, TOTAL = 8.5, 10, 12.5, 13, 17, 20, 34

W, H = A4
M, G, R = 48, 8, 8          # margin, gutter, module radius (one radius everywhere)
CW = W - 2 * M
ROW_H, LEAD = 52, 15
LIMIT = 650                 # rows end above this so the total window + footer always fit


def tw(s, font, size, hs=100, cs=0):
    return pdfmetrics.stringWidth(s, font, size) * hs / 100 + cs * hs / 100 * len(s)


def text(c, x, y, s, font, size, color, anchor="l", hs=100, cs=0):
    w = tw(s, font, size, hs, cs)
    if anchor == "r":
        x -= w
    elif anchor == "c":
        x -= w / 2
    t = c.beginText(x, H - y)
    t.setFont(font, size)
    t.setHorizScale(hs)
    t.setCharSpace(cs)
    c.setFillColor(color)
    t.textOut(s)
    t.setCharSpace(0)
    t.setHorizScale(100)
    c.drawText(t)


def legend(c, x, y, s, color=GRAY, anchor="l"):
    text(c, x, y, s, "SB", LBL, color, anchor=anchor, cs=1.6)


def rrect(c, x, y, w, h, r, fill):
    c.setFillColor(fill)
    c.roundRect(x, H - y - h, w, h, r, fill=1, stroke=0)


def line(c, x0, x1, y, color, lw=0.8):
    c.setStrokeColor(color)
    c.setLineWidth(lw)
    c.line(x0, H - y, x1, H - y)


def money(v):
    whole, frac = f"{v:.2f}".split(".")
    if len(whole) > 3:
        head, tail, parts = whole[:-3], whole[-3:], []
        while len(head) > 2:
            parts.insert(0, head[-2:])
            head = head[:-2]
        if head:
            parts.insert(0, head)
        whole = ",".join(parts + [tail])
    return f"\u20b9{whole}.{frac}"


def wrap(s, font, size, maxw):
    lines, cur = [], ""
    for w in s.split():
        t = (cur + " " + w).strip()
        if tw(t, font, size) <= maxw:
            cur = t
        else:
            if cur:
                lines.append(cur)
            cur = w
    if cur:
        lines.append(cur)
    return lines or [s]


def footer(c):
    line(c, M, W - M, H - 66, HAIR)
    text(c, M, H - 44, FOOTER, "S", 9, GRAY)


def brand_module(c):
    rrect(c, M, 48, 335, 92, R, MOD)
    size = WORD
    while tw(NAME, "SB", size, 118, 1.2) > 335 - 36 and size > 12:
        size -= 0.5
    text(c, M + 18, 86, NAME, "SB", size, INK, hs=118, cs=1.2)
    y = 108
    for ln in BUSINESS_LINES:
        text(c, M + 18, y, ln, "S", BODY, GRAY)
        y += 14
    dx = M + 335 + G
    rrect(c, dx, 48, CW - 335 - G, 92, R, MOD)
    legend(c, dx + 16, 78, "DATE")
    text(c, dx + 16, 108, DATE, "SB", DATE_V, INK)


def billed_module(c):
    rrect(c, M, 148, CW, 68, R, MOD)
    legend(c, M + 18, 174, "BILLED TO")
    size = REC
    while tw(BILL_TO, "SB", size) > CW - 36 and size > 11:
        size -= 0.5
    text(c, M + 18, 199, BILL_TO, "SB", size, INK)


def table_header(c, top):
    legend(c, M + 30, top + 22, "NO.", anchor="c")
    legend(c, M + 58, top + 22, "ITEM")
    legend(c, W - M - 18, top + 22, "PRICE", anchor="r")
    sy = top + 31                                   # dial-scale ruler
    line(c, M + 18, W - M - 18, sy, TICK, 0.7)
    x, i = M + 18, 0
    c.setStrokeColor(TICK)
    c.setLineWidth(0.7)
    while x <= W - M - 18 + 0.01:
        ln = 7 if i % 5 == 0 else 3.6
        c.line(x, H - sy, x, H - sy - ln)
        x += 8
        i += 1


def total_window(c, y, amt):
    wx, ww, wh = W - M - 300, 300, 84
    rrect(c, wx, y, ww, wh, 8, CHAR)
    c.setStrokeColor(BEZEL)
    c.setLineWidth(0.8)
    c.roundRect(wx + 3, H - y - wh + 3, ww - 6, wh - 6, 6, fill=0, stroke=1)
    legend(c, wx + 22, y + 28, "TOTAL", color=DLEG)
    c.setFillColor(ORANGE)                          # the indicator light
    c.circle(wx + 22 + tw("TOTAL", "SB", LBL, 100, 1.6) + 9, H - (y + 25), 2.6, fill=1, stroke=0)
    size = TOTAL
    while pdfmetrics.stringWidth(amt, "M", size) > ww - 44 and size > 14:
        size -= 0.5
    text(c, wx + ww - 22, y + 66, amt, "M", size, ORANGE, anchor="r")


def build():
    # paginate rows
    rows = []
    for i, (item, price) in enumerate(ITEMS, 1):
        pw = tw(money(price), "SB", ROW)
        lines = wrap(item, "S", ROW, (W - M - 18) - pw - 24 - (M + 58))
        rows.append(dict(i=i, lines=lines, price=price, rh=max(ROW_H, 38 + LEAD * (len(lines) - 1))))
    pages, cur, y = [], [], 224 + 40
    for r in rows:
        if y + r["rh"] > LIMIT and cur:
            pages.append(cur)
            cur, y = [], 100 + 40
        cur.append(r)
        y += r["rh"]
    pages.append(cur)

    c = canvas.Canvas(OUT, pagesize=A4)
    c.setTitle(f"Invoice - {BILL_TO}")
    c.setAuthor("Viva Electronics")
    for p, page_rows in enumerate(pages):
        if p == 0:
            brand_module(c)
            billed_module(c)
            top = 224
        else:
            rrect(c, M, 48, CW, 44, R, MOD)
            text(c, M + 18, 76, NAME, "SB", 15, INK, hs=118, cs=1.2)
            top = 100
        mh = 40 + sum(r["rh"] for r in page_rows) + 8
        rrect(c, M, top, CW, mh, R, MOD)
        table_header(c, top)
        y = top + 40
        for k, r in enumerate(page_rows):
            mid = y + r["rh"] / 2
            c.setStrokeColor(INK)
            c.setLineWidth(1)
            c.circle(M + 30, H - mid, 11, fill=0, stroke=1)
            text(c, M + 30, mid + 3.5, f"{r['i']:02d}", "SB", 9.5, INK, anchor="c")
            n = len(r["lines"])
            fb = mid + 4.5 - LEAD * (n - 1) / 2
            for j, ln in enumerate(r["lines"]):
                text(c, M + 58, fb + LEAD * j, ln, "S", ROW, INK)
            text(c, W - M - 18, mid + 4.5, money(r["price"]), "SB", ROW, INK, anchor="r")
            y += r["rh"]
            if k < len(page_rows) - 1:
                line(c, M + 18, W - M - 18, y, white, 1.6)
        if p == len(pages) - 1:
            total_window(c, top + mh + G, money(sum(pr for _, pr in ITEMS)))
        footer(c)
        c.showPage()
    c.save()
    print("saved", OUT)


build()

pdfmetrics.registerFont(TTFont("JS", FONT_DIR + "JosefinSans-SemiBold.ttf"))

# ---------- Palette: emerald-black, antique gold, ivory ----------
INK, GOLD, GOLDL = HexColor("#0B2B26"), HexColor("#B08D3E"), HexColor("#D2AE5E")
IVORY, CREAM, DIM = HexColor("#F4EEDD"), HexColor("#F7F2E6"), HexColor("#BFC7BE")

# ---------- Type scale (pt) ----------

WORD_CS, LBL_CS = 4, 2

W, H = A4
M = 48
CW = W - 2 * M
LX, RX = 64, W - M - 16          # left / right text edges shared by every section
ROW_H, LEAD, LIMIT = 56, 15, 640


# ---------- helpers (y is measured DOWN from the top edge) ----------
def tw(s, font, size, cs=0):
    return pdfmetrics.stringWidth(s, font, size) + cs * len(s)


def text(c, x, y, s, font, size, color, anchor="l", cs=0):
    w = tw(s, font, size, cs)
    if anchor == "r":
        x -= w
    elif anchor == "c":
        x -= w / 2
    t = c.beginText(x, H - y)
    t.setFont(font, size)
    t.setCharSpace(cs)
    c.setFillColor(color)
    t.textOut(s)
    t.setCharSpace(0)
    c.drawText(t)


def legend(c, x, y, s, color, anchor="l"):
    text(c, x, y, s, "JS", LBL, color, anchor=anchor, cs=LBL_CS)


def money(v):
    whole, frac = f"{v:.2f}".split(".")
    if len(whole) > 3:
        head, tail, parts = whole[:-3], whole[-3:], []
        while len(head) > 2:
            parts.insert(0, head[-2:])
            head = head[:-2]
        if head:
            parts.insert(0, head)
        whole = ",".join(parts + [tail])
    return f"\u20b9{whole}.{frac}"


def wrap(s, font, size, maxw):
    lines, cur = [], ""
    for w in s.split():
        t = (cur + " " + w).strip()
        if tw(t, font, size) <= maxw:
            cur = t
        else:
            if cur:
                lines.append(cur)
            cur = w
    if cur:
        lines.append(cur)
    return lines or [s]


def stepped(x, y, w, h, s, n):
    """Rectangle whose four corners are cut as n-step staircases of size s."""
    p = [(x + n * s, y)]
    for k in range(n):
        p += [(x + w - (n - 1 - k) * s, y + k * s), (x + w - (n - 1 - k) * s, y + (k + 1) * s)]
    for k in range(n):
        p += [(x + w - k * s, y + h - (n - 1 - k) * s), (x + w - (k + 1) * s, y + h - (n - 1 - k) * s)]
    for k in range(n):
        p += [(x + n * s - (k + 1) * s, y + h - k * s), (x + n * s - (k + 1) * s, y + h - (k + 1) * s)]
    for k in range(n):
        p += [(x + k * s, y + n * s - (k + 1) * s), (x + (k + 1) * s, y + n * s - (k + 1) * s)]
    return p


def plinth(x, y, w, h, d=0):
    """Ziggurat silhouette: widest at the base, two setbacks toward the top. d = inward offset."""
    return [(x + d, y + h - d), (x + d, y + h - 28 + d), (x + 6 + d, y + h - 28 + d),
            (x + 6 + d, y + h - 56 + d), (x + 12 + d, y + h - 56 + d), (x + 12 + d, y + d),
            (x + w - 12 - d, y + d), (x + w - 12 - d, y + h - 56 + d), (x + w - 6 - d, y + h - 56 + d),
            (x + w - 6 - d, y + h - 28 + d), (x + w - d, y + h - 28 + d), (x + w - d, y + h - d)]


def shape(c, pts, fill=None, stroke=None, lw=0.8):
    p = c.beginPath()
    p.moveTo(pts[0][0], H - pts[0][1])
    for x, y in pts[1:]:
        p.lineTo(x, H - y)
    p.close()
    if fill is not None:
        c.setFillColor(fill)
    if stroke is not None:
        c.setStrokeColor(stroke)
        c.setLineWidth(lw)
    c.drawPath(p, fill=1 if fill is not None else 0, stroke=1 if stroke is not None else 0)


def hline(c, y, color, lw=0.7, x0=M, x1=W - M):
    c.setStrokeColor(color)
    c.setLineWidth(lw)
    c.line(x0, H - y, x1, H - y)


def double_rule(c, y):
    hline(c, y, GOLD, 1.4)
    hline(c, y + 3.5, GOLD, 0.6)


def frame(c):
    shape(c, stepped(24, 24, W - 48, H - 48, 8, 3), stroke=GOLD, lw=1.4)
    shape(c, stepped(30, 30, W - 60, H - 60, 8, 3), stroke=INK, lw=0.6)


def footer(c):
    tws = tw(FOOTER, "JR", FOOT)
    cx, by = W / 2, H - 48
    text(c, cx, by, FOOTER, "JR", FOOT, GRAY, anchor="c")
    hline(c, by - 3.5, GOLD, 0.6, M, cx - tws / 2 - 16)
    hline(c, by - 3.5, GOLD, 0.6, cx + tws / 2 + 16, W - M)


def band(c, h):
    shape(c, stepped(M, 48, CW, h, 6, 2), fill=INK)
    shape(c, stepped(M + 4, 52, CW - 8, h - 8, 6, 2), stroke=GOLDL, lw=0.8)


def first_page_top(c):
    band(c, 104)
    size = WORD
    while tw(NAME, "PO", size, WORD_CS) > 320 and size > 14:
        size -= 0.5
    text(c, LX, 92, NAME, "PO", size, CREAM, cs=WORD_CS)
    y = 118
    for ln in BUSINESS_LINES:
        text(c, LX, y, ln, "JR", DET, DIM)
        y += 15
    legend(c, RX, 92, "DATE", GOLDL, anchor="r")
    text(c, RX, 118, DATE, "JS", DATE_V, CREAM, anchor="r")

    # billed to: a three-line "pilaster" instead of a box
    for dx, lw in ((0, 1.4), (4, 0.7), (7.5, 0.7)):
        c.setStrokeColor(GOLD)
        c.setLineWidth(lw)
        c.line(M + dx, H - 188, M + dx, H - 234)
    legend(c, LX, 202, "BILLED TO", GRAY)
    name = BILL_TO.upper()
    size = REC
    while tw(name, "JS", size, 1.5) > RX - LX and size > 11:
        size -= 0.5
    text(c, LX, 226, name, "JS", size, INK, cs=1.5)
    return 270


def table_header(c, top):
    double_rule(c, top)
    shape(c, stepped(M, top + 11, CW, 26, 4, 2), fill=IVORY)
    legend(c, M + 29, top + 28, "NO.", INK, anchor="c")
    legend(c, LX + 44, top + 28, "ITEM", INK)
    legend(c, RX, top + 28, "PRICE", INK, anchor="r")
    return top + 45


def total_plinth(c, y, amt):
    x, w, h = W - M - 300, 300, 84
    shape(c, plinth(x, y, w, h), fill=INK)
    shape(c, plinth(x, y, w, h, 4), stroke=GOLDL, lw=0.8)
    legend(c, x + 28, y + 30, "TOTAL", GOLDL)
    size = TOTAL
    while tw(amt, "JS", size) > RX - (x + 28) - 12 and size > 14:
        size -= 0.5
    text(c, RX, y + 66, amt, "JS", size, CREAM, anchor="r")


def build():
    rows = []
    for i, (item, price) in enumerate(ITEMS, 1):
        pw = tw(money(price), "JS", ROW)
        lines = wrap(item, "JR", ROW, RX - pw - 24 - (LX + 44))
        rows.append(dict(i=i, lines=lines, price=price, rh=max(ROW_H, 38 + LEAD * (len(lines) - 1))))
    pages, cur, y = [], [], 270 + 45
    for r in rows:
        if y + r["rh"] > LIMIT and cur:
            pages.append(cur)
            cur, y = [], 128 + 45
        cur.append(r)
        y += r["rh"]
    pages.append(cur)

    c = canvas.Canvas(OUT, pagesize=A4)
    c.setTitle(f"Invoice - {BILL_TO}")
    c.setAuthor("Viva Electronics")
    for p, page_rows in enumerate(pages):
        frame(c)
        if p == 0:
            top = first_page_top(c)
        else:
            band(c, 56)
            text(c, LX, 86, NAME, "PO", 20, CREAM, cs=3)
            top = 128
        y = table_header(c, top)
        for k, r in enumerate(page_rows):
            mid = y + r["rh"] / 2
            shape(c, [(M + 29, mid - 13), (M + 42, mid), (M + 29, mid + 13), (M + 16, mid)], stroke=GOLD, lw=1)
            text(c, M + 29, mid + 3.4, f"{r['i']:02d}", "JS", 9.5, INK, anchor="c")
            n = len(r["lines"])
            fb = mid + 4.5 - LEAD * (n - 1) / 2
            for j, ln in enumerate(r["lines"]):
                text(c, LX + 44, fb + LEAD * j, ln, "JR", ROW, INK)
            text(c, RX, mid + 4.5, money(r["price"]), "JS", ROW, INK, anchor="r")
            y += r["rh"]
            if k < len(page_rows) - 1:
                hline(c, y, HAIR, 0.7)
        double_rule(c, y)
        if p == len(pages) - 1:
            total_plinth(c, y + 3.5 + 28, money(sum(pr for _, pr in ITEMS)))
        footer(c)
        c.showPage()
    c.save()
    print("saved", OUT)


build()

def poly(c, pts, fill):

# ---------- sheet frame: border, grid-reference band with zone ticks (no letters), centring marks ----------
def frame(c):
    box(c, 28, 28, W - 56, H - 56, 1.4)
    box(c, 34, 34, W - 68, H - 68, 0.5)
    for k in range(1, 4):                                   # 4 horizontal zones
        x = 28 + (W - 56) * k / 4
        ln(c, x, 28, x, 34, 0.5)
        ln(c, x, H - 34, x, H - 28, 0.5)
    for k in range(1, 6):                                   # 6 vertical zones
        y = 28 + (H - 56) * k / 6
        ln(c, 28, y, 34, y, 0.5)
        ln(c, W - 34, y, W - 28, y, 0.5)
    ln(c, W / 2, 10, W / 2, 40, 1.2)                        # centring marks
    ln(c, W / 2, H - 40, W / 2, H - 10, 1.2)
    ln(c, 10, H / 2, 40, H / 2, 1.2)
    ln(c, W - 40, H / 2, W - 10, H / 2, 1.2)

def dimension(c, yb, amt):
    xr = W - M
    xl = C2
    size = TOTAL
    arrow, pad = 9, 10
    avail = (xr - xl) - 2 * (arrow + pad) - 16
    while tw(amt, "MS", size) > avail and size > 18:
        size -= 0.5
    need = tw(amt, "MS", size) + 2 * (arrow + pad) + 16
    if need > xr - xl:
        xl = xr - need
    yd = yb + 38
    for x in (xl, xr):                                         # extension lines: small gap, run past the dimension line
        ln(c, x, yb + 3, x, yd + 5, 0.6, DIMC)
    xc = (xl + xr) / 2
    half = tw(amt, "MS", size) / 2 + 8
    ln(c, xl, yd, xc - half, yd, 0.6, DIMC)                    # dimension line, broken for the figure
    ln(c, xc + half, yd, xr, yd, 0.6, DIMC)
    poly(c, [(xl, yd), (xl + arrow, yd - 1.7), (xl + arrow, yd + 1.7)], DIMC)
    poly(c, [(xr, yd), (xr - arrow, yd - 1.7), (xr - arrow, yd + 1.7)], DIMC)
    text(c, xc, yd + 0.35 * size, amt, "MS", size, DIMC, anchor="c")
    legend(c, xl - 14, yd + 3, "TOTAL", GRAY, anchor="r")
